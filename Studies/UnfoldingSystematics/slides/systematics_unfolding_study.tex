\begin{frame}{Run Configuration}

  \item Unfolding engine: \texttt{Unfolding/unfold\_thrust.py}
  \item Plotting script: \texttt{Unfolding/plot\_unfolded.py}
  \item Command used:
\end{itemize}
\vspace{0.2em}
\texttt{python3 Studies/UnfoldingSystematics/scripts/run\_systematic\_unfolding.py --allow-missing-cut-data}
\vspace{0.4em}
\begin{itemize}
  \item Variation logic follows \texttt{../unfold-ee-logtau/run\_ibu.py}:
  \item Cut-based variations vary data + reco MC.
  \item Neutral-response variations vary reco MC only (nominal data kept).
\end{itemize}
