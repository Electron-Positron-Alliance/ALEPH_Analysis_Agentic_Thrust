\title{Analysis note: precision QCD as a testing ground for precision AI, an agentic measurement of thrust with ALEPH archived data}

\author[a]{Anthony Badea}
\affiliation[a]{University of Chicago, Enrico Fermi Institute, 60637 IL, USA}
\emailAdd{anthony.badea@cern.ch}

\abstract{
We present a measurement of the thrust distribution in $e^{+}e^{-}$ collisions at $\sqrt{s}=91.2$~GeV using archived ALEPH data, carried out entirely by an AI agent under expert physicist direction---no analysis code was written by hand. 
A detector-corrected spectrum with full covariance is obtained via Iterative Bayesian Unfolding, suitable for precision QCD phenomenology. 
This work is a first step toward a closed theory--experiment loop in which AI agents perform rigorous measurements directly confrontable with AI-assisted theoretical calculations, accelerating the cycle that drives discovery in fundamental physics. 
We argue that precision QCD, leveraging the open LEP data and advanced theoretical calculations, provides an ideal testing ground for developing such precision AI systems.
}

\keywords{QCD, event shapes, electron-positron collisions, unfolding}
